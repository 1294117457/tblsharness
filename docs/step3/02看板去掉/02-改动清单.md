### 2.1 `harness.db.showOnCanvas`

**位置**：`src/commands/canvas.ts`（与现有 `harness.source.addToCanvas` 一起）

**改动**：

```typescript
/** Shows a db table on the root canvas without copying it >  to the design table. The table stays a db table; the > user can read it side-by-side >  with the design. */
register(h, 'harness.db.showOnCanvas', async (arg) => {
  const { ws, kind, ref, table } = await sourceFromArg(h, arg);
  if (kind !== 'db') throw new Error('只能把数据库表加到画布');
  if (!table) throw new Error('请选择一张数据库表');
  const design = await pickSourceId(h, ws, 'design', { workspace: ws.id });
  const meta = await ws.design(design).readMeta();
  // 确保 db 在 design 的 sources 里
  if (!meta.sources?.includes(ref)) {
    await ws.design(design).writeMeta({ ...meta, sources: [...(meta.sources ?? []), ref] });
    h.store.invalidate({ workspace: ws.id, kind: 'design', id: design });
  }
  const layout = await h.canvases.layout(ws.id, design);
  if (!layout.nodes.some((n) => n.source === ref && n.table === table)) {
    const root = layout.nodes.filter((n) => !n.partition);
    const x = root.length ? Math.max(...root.map((n) => n.x)) + 320 : 0;
    await h.canvases.editLayout(ws.id, design, `添加表 ${table}`, [{ op: 'nodes.put', nodes: [{ source: ref, table, x, y: 0 }] }]);
  }
  await h.canvases.revealTable(ws.id, design, ref, table);
});
```

### 2.3 `harness.db.addToDesign`

**位置**：`src/commands/canvas.ts`

**改动**：

```typescript
/** Adds a database to a design's references without putting any tables on the canvas. The
 * >  workspace must have >  already hanned the db >  (`workspace.db(dbId)`). */
register(h, 'harness.db.addToDesign', async (arg) => {
  const ws = await pickWorkspace(h, arg);
  if (!arg?.kind || arg.kind !== 'db' || !arg.id) throw new Error('请在侧边栏选择一个数据库');
  const dbId = arg.id;
  const design = await pickSourceId(h, ws, 'design', { workspace: ws.id });
  const meta = await ws.design(design).readMeta();
  if (meta.sources?.includes(dbId)) {
    vscode.window.setStatusBarMessage('Harness：这个数据库已经在设计画布里了', 2000);
    return;
  }
  await ws.design(design).writeMeta({ ...meta, sources: [...(meta.sources ?? []), dbId] });
  h.store.invalidate({ workspace: ws.id, kind: 'design', id: design });
  // 不主动开画布，让用户自己决定。如果画布已经开着，store invalidate 会推 sources。
});
```

**注意**：这条命令**不**通过 `canvasEditor.addDbSource`（webview 协议），而**直接**写 `meta.sources`。`store.invalidate` 会让所有打开的画布重新推 sources（`pushSources` in `canvasEditor.ts`）。
